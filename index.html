    <h2>WU Check-in</h2>

    <p>ลงทะเบียนเข้าพื้นที่ ม.วลัยลักษณ์</p>

    

    <input type="text" id="empId" placeholder="รหัสประจำตัว / ID">

    <input type="text" id="empName" placeholder="ชื่อ-นามสกุล">

    

    <button id="btnSubmit" onclick="startProcess()">บันทึกข้อมูล</button>

<script>

    // ==========================================

    // ⚙️ ตั้งค่าตำแหน่ง (Configuration)

    // ==========================================

    

    // พิกัดมหาวิทยาลัยวลัยลักษณ์ (จุดกลาง)

    const TARGET_LAT = 8.641774; 

    const TARGET_LNG = 99.897022; 

    

    // รัศมีที่อนุญาต (เมตร) : 1000 เมตร = 1 กิโลเมตร (ครอบคลุมพื้นที่ส่วนใหญ่)

    const RADIUS_LIMIT = 1000; 


    // Web App URL ของคุณ

    const SCRIPT_URL = "https://script.google.com/macros/s/AKfycbzcL9cHZ95csBD9xoMcGwTQ_cIoPG6RLwRsrjcvacEfU1Xgkdk7yhuxTVCn49zSye3Y/exec";


    // ==========================================


    function startProcess() {

        const id = document.getElementById('empId').value;

        const name = document.getElementById('empName').value;

        const btn = document.getElementById('btnSubmit');


        if (!id || !name) {

            alert("กรุณากรอก ID และ ชื่อ ให้ครบถ้วน");

            return;

        }


        btn.disabled = true;

        btn.innerText = "กำลังดำเนินการ...";

        updateStatus("กำลังตรวจเช็คตำแหน่ง GPS...", "#3498db");


        if (navigator.geolocation) {

            navigator.geolocation.getCurrentPosition(position => {

                const userLat = position.coords.latitude;

                const userLng = position.coords.longitude;

                

                const distance = getDistance(userLat, userLng, TARGET_LAT, TARGET_LNG);

                

                let statusMsg = (distance <= RADIUS_LIMIT) ? "ในพื้นที่" : "นอกพื้นที่";

                let color = (distance <= RADIUS_LIMIT) ? "#27ae60" : "#c0392b";


                updateStatus(`สถานะ: ${statusMsg} (${Math.round(distance)} ม. จากจุดกลาง)`, color);

                

                // ส่งข้อมูลไปที่ Google Sheets

                sendData(id, name, userLat, userLng, statusMsg);


            }, (error) => {

                updateStatus("ผิดพลาด: โปรดเปิด GPS และกดยอมรับการเข้าถึงตำแหน่ง", "#c0392b");

                btn.disabled = false;

                btn.innerText = "ลองอีกครั้ง";

            }, { enableHighAccuracy: true });

        } else {

            updateStatus("เบราว์เซอร์ไม่รองรับ GPS", "#c0392b");

        }

    }


    function updateStatus(text, color) {

        const box = document.getElementById('status-box');

        box.style.display = "block";

        box.innerText = text;

        box.style.color = color;

        box.style.border = `1px solid ${color}`;

        box.style.backgroundColor = color + "11";

    }


    function getDistance(lat1, lon1, lat2, lon2) {

        const R = 6371000;

        const dLat = (lat2 - lat1) * Math.PI / 180;

        const dLon = (lon2 - lon1) * Math.PI / 180;

        const a = Math.sin(dLat/2) * Math.sin(dLat/2) +

                  Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) *

                  Math.sin(dLon/2) * Math.sin(dLon/2);

        return R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));

    }


    function sendData(id, name, lat, lng, status) {

        fetch(SCRIPT_URL, {

            method: 'POST',

            mode: 'no-cors',

            body: JSON.stringify({ id, name, lat, lng, status })

        }).then(() => {

            updateStatus("✅ บันทึกข้อมูลลง Google Sheets เรียบร้อยแล้ว", "#27ae60");

            document.getElementById('btnSubmit').innerText = "บันทึกสำเร็จ";

        }).catch(err => {

            updateStatus("เกิดข้อผิดพลาดในการส่งข้อมูล", "#c0392b");

            document.getElementById('btnSubmit').disabled = false;

        });

    }
